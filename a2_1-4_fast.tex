\vspace{2pt}
{\small\renewcommand{\arraystretch}{1.15}%
\begin{tabularx}{\textwidth}{|l|Y|}
\hline
\textbf{arithmetic sequence} (secuencia aritm\'etica) & constant difference between consecutive terms: the \textbf{common difference} $d$ (diferencia com\'un). $3, 8, 13, 18, \ldots$ has $d=5$. \\ \hline
\textbf{recursive definition} (definici\'on recursiva) & each term is defined by operations on the \textbf{previous term}: $a_1$ is given, then $a_n=a_{n-1}+d$ for $n>1$ \\ \hline
\textbf{explicit definition} (definici\'on expl\'icita) & finds any term from its \textbf{term number} $n$ without the previous term: $a_n=a_1+d(n-1)$ \\ \hline
\end{tabularx}}

\wspart{Example 3}{Translate Between Recursive and Explicit Forms}
\textbf{A.} Given the recursive definition $a_n=\begin{cases}3, & n=1\\ a_{n-1}+0.5, & n>1\end{cases}$, what is an explicit definition for the sequence?

\begin{minipage}[t]{0.53\textwidth}
\vspace{0pt}
Use the recursive definition to find a pattern:\\[2pt]
$a_1=3$\\
$a_2=a_1+0.5=3+0.5$\\
$a_3=a_2+0.5=[3+0.5]+0.5=3+2(0.5)$\\
$a_4=a_3+0.5=[3+2(0.5)]+0.5=3+3(0.5)$\\[2pt]
Term $n$ has $(n-1)$ common differences added to $a_1$.
\end{minipage}\hfill
\begin{minipage}[t]{0.43\textwidth}
\vspace{0pt}
\small\renewcommand{\arraystretch}{1.15}%
\begin{tabular}{|c|l|c|}
\hline
\textbf{term no.} $n$ & \textbf{built from $a_1$} & \textbf{term} \\ \hline
1 & $3+0(0.5)$ & 3 \\
2 & $3+1(0.5)$ & 3.5 \\
3 & $3+2(0.5)$ & 4 \\
4 & $3+3(0.5)$ & 4.5 \\
$n$ & $3+(n-1)(0.5)$ & \\ \hline
\end{tabular}
\end{minipage}

\vspace{4pt}
Explicit definition: \fbox{$a_n=3+(n-1)(0.5)$}. Check: $a_4=3+3(0.5)=4.5$ \checkmark

\textbf{B.} Given the explicit definition $a_n=16-3(n-1)$, what is the recursive definition for the arithmetic sequence?

\begin{minipage}[t]{0.56\textwidth}
\vspace{0pt}
\begin{enumerate}[label=\arabic*.]
\item Match $a_n=a_1+d(n-1)$: the number multiplying $(n-1)$ is the common difference, $d=-3$.
\item First term: $a_1=16-3(1-1)=16$.
\item Check: $a_2=16-3(2-1)=13$, and $16-3=13$ \checkmark.
\end{enumerate}
\end{minipage}\hfill
\begin{minipage}[t]{0.40\textwidth}
\vspace{0pt}
Recursive definition:\\[3pt]
\fbox{$a_n=\begin{cases}16, & n=1\\ a_{n-1}-3, & n>1\end{cases}$}
\end{minipage}

\vspace{4pt}
