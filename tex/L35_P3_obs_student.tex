\begin{center}
\begin{tikzpicture}[scale=0.85]
  \draw[thin,gray!40] (-4,-3) grid (5,2);
  \draw[->] (-4.4,0) -- (5.4,0) node[right] {\small $x$};
  \draw[->] (0,-3.4) -- (0,2.4) node[above] {\small $y$};
  \foreach \x in {-4,-3,-2,-1,1,2,3,4,5} \node[below,font=\scriptsize] at (\x,-0.05) {$\x$};
  % ends both down, touches at -1, crosses at 3 -> even-degree polynomial, negative leading coeff, double root at -1, simple root at 3 PLUS another simple root somewhere left to keep ends-down. Or quartic with double at -1 and... we'll use degree-4: -(x+1)^2*(x-3)*(x-a). For symmetry pick a = -3 just to show ends go down.
  \draw[thick, blue, domain=-3.5:4, samples=200, smooth] plot (\x, {-0.10*(\x+1)*(\x+1)*(\x-3)*(\x+3.5)});
  \node[font=\tiny] at (-1,0.3) {touch};
  \node[font=\tiny] at (3,0.3) {cross};
