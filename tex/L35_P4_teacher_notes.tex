  \item No phones. Pencil only. Clear desk.
  \item Start timer visibly.
}%
}{%
\phaseitems{
  \item Clear desk except pencil + calculator.
  \item Raise hand if a question is ambiguous (wording, not math).
}%
}{%
\phaseitems{
  \item Any logistical questions before we start?
}%
}{Solo teacher: reset room tone. Start timer visibly.}

\phasetag{proctoring \textperiodcentered\ silent}
\frameworkphaseheader{Assessment (Topic 3)}{1--3 mix}{35}{%
\phaseitems{
  \item Circulate silently every 10 min.
  \item At minute 20: ``15 min remaining. Flag anything you need to skip.''
  \item At minute 30: ``5 min. Finalize your answers.''
}%
}{%
\phaseitems{
  \item Work independently. No partner talk.
  \item If stuck \(>\)2 min on a question, flag and move on.
}%
}{%
\phaseitems{
  \item \textit{(Silent proctoring --- no teaching during assessment.)}
}%
}{Solo teacher: proctor, time, say nothing content-related.}
