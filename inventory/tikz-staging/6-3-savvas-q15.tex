\usepackage{tikz}
\usepackage{pgfplots}

\begin{document}

    grid style={line width=.1pt, draw=gray!30},
    axis line style={-Stealth},
    legend style={font=\small},
    tick label style={font=\small},
    xmin=-0.3, xmax=4.5, ymin=-5, ymax=100,
    xtick={0,1,2,3,4},
