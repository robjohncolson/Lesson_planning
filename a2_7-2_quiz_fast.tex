\noindent{\LARGE\textbf{7-2 Lesson Quiz, short form: answer key}}

\vspace{4pt}\noindent\rule{\textwidth}{0.6pt}

{\raggedright
\begin{enumerate}
\item[\textbf{1.}] \textbf{$\sin\frac{2\pi}{3}=\frac{\sqrt3}{2}$, \ $\cos\frac{2\pi}{3}=-\frac12$.} Quadrant II, reference angle $\pi-\frac{2\pi}{3}=\frac\pi3$; terminal point $\left(-\frac12,\frac{\sqrt3}{2}\right)$. Distractors: $\frac12$ and $-\frac{\sqrt3}{2}$ come from using the wrong leg (reference angle $\frac\pi6$ instead of $\frac\pi3$, the $\frac{5\pi}{6}$ point); $\sqrt3$ is $|\tan\frac{2\pi}{3}|$. Savvas quiz item 1, DOK 1, Skills Review and Practice F84. Taught by Example 4C (the same angle), so this item is a recall check of the packet. \textbf{10 points:} 5 per correct box.
\item[\textbf{3.}] \textbf{A, D, E.} A: $-\frac{4\pi}{3}+2\pi=\frac{2\pi}{3}$ (Quadrant II), $\tan\frac{2\pi}{3}=-\sqrt3$, true. B: $-\frac{7\pi}{4}+2\pi=\frac\pi4$, $\cos\frac\pi4=+\frac{\sqrt2}{2}$, false. C: $-\frac{11\pi}{3}+4\pi=\frac\pi3$, $\cot\frac\pi3=+\frac{\sqrt3}{3}$, false. D: $\frac{7\pi}{6}$ is in Quadrant III with reference angle $\frac\pi6$, $\cos\frac{7\pi}{6}=-\frac{\sqrt3}{2}$, $\sec\frac{7\pi}{6}=-\frac{2}{\sqrt3}=-\frac{2\sqrt3}{3}$, true. E: $\frac{13\pi}{6}-2\pi=\frac\pi6$, $\sin\frac\pi6=\frac12$, true. Savvas quiz item 3, DOK 1, F84. Taught by Examples 4 and 5; the coterminal step (adding or subtracting $2\pi$) is Lesson 7-1, and Try It 4b ($-210^\circ$) is the packet's only negative angle. \textbf{Bonus 5:} A, D and E and no other box; 3 for two of the three and no wrong box; 0 otherwise.
\item[\textbf{4.}] \textbf{A, C, E.} $225^\circ$ is in Quadrant III with reference angle $45^\circ$; terminal point $\left(-\frac{\sqrt2}{2},-\frac{\sqrt2}{2}\right)$. \mbox{$\csc225^\circ=\frac{1}{-\sqrt2/2}=-\frac{2}{\sqrt2}=-\sqrt2$}; $\sec225^\circ=-\sqrt2$; $\cot225^\circ=\frac{x}{y}=1$. B, D and F are false (F has the wrong sign). Savvas quiz item 4, DOK 1, F84. Taught by Example 5 with Example 4's Quadrant III signs; the packet Practice ($135^\circ$, TE Practice 14 and Do You Know How 11) is the parallel. \textbf{10 points:} all three boxes and no wrong box 10; all three with one wrong box, or two correct and none wrong, 6; one correct and none wrong 3; 0 otherwise.
\end{enumerate}
}
