\packethead{Lesson 1-5}{Solving by Graphing}

\goals{I can solve an equation or inequality by graphing both sides and reading where the graphs meet.}{I can say ``the solution is where the graphs intersect'' and ``the inequality is true where one graph is above (or below) the other.''}

\quiztarget{You will (1) sketch two graphs to solve an equation and read the solution(s) off the intersection, (2) graph two cost equations and write the inequality for when one plan is cheaper, and (3) approximate where two functions are equal from a table of values.}

\wspart{Example 1}{Use a Graph to Solve an Equation}
\textbf{The idea:} graph $y=$ left side and $y=$ right side. The $x$-value of each \textbf{intersection} is a solution.

\begin{minipage}[t]{0.47\textwidth}
\textbf{A.} Solve $-3x+20=0.2x+4$ by graphing.
\begin{enumerate}[label=\arabic*.]
\item Graph $y=-3x+20$ and $y=0.2x+4$.
\item Find the intersection: $(5,5)$.
\item Solution: $x=5$. Check: $-15+20=5$ and $1+4=5$. \checkmark
\end{enumerate}

\begin{axis}[sheetgrid, xmin=-5.5, xmax=10.5, ymin=-5.5, ymax=25.5,
  xtick={-4,-2,...,10}, ytick={-5,0,...,25}, x=0.38cm, y=0.2cm]
\addplot[very thick, blue] coordinates {(-1.9,25.7) (8.6,-5.8)};
\addplot[very thick, red] coordinates {(-5.5,2.9) (10.5,6.1)};
\node[closed] at (axis cs:5,5) {};
\node[font=\scriptsize, fill=white, inner sep=1pt, anchor=west] at (axis cs:5.4,6.5) {$(5,5)$};
\node[font=\scriptsize, blue, fill=white, inner sep=1pt, anchor=west] at (axis cs:-4.8,18) {$y=-3x+20$};
\node[font=\scriptsize, red, fill=white, inner sep=1pt] at (axis cs:3,12) {$y=0.2x+4$};
\end{axis}
\end{tikzpicture}
\end{center}
\end{minipage}\hfill
\begin{minipage}[t]{0.50\textwidth}
\textbf{B.} Solve $|x-4|=\tfrac12x+1$ by graphing.
\begin{enumerate}[label=\arabic*.]
\item Graph $y=|x-4|$ (a V, vertex $(4,0)$) and $y=\tfrac12x+1$.
\item Two intersections: $(2,2)$ and $(10,6)$.
\item Solutions: $x=2$ and $x=10$. Check: $|2-4|=2$ and $\tfrac12(2)+1=2$. \checkmark
\end{enumerate}

\wspart{Example 2}{Solve a One-Variable Inequality by Graphing}
\textbf{The idea:} graph both sides. ``Left $>$ right'' is true where the left graph is \textbf{above} the right one; intersections are the boundaries.

\begin{minipage}[t]{0.42\textwidth}
\textbf{A.} Solve $x^2-4>0$.
\begin{enumerate}[label=\arabic*.]
\item Graph $y=x^2-4$ and $y=0$ (the $x$-axis).
\item Boundaries: the graph crosses the $x$-axis at $x=-2$ and $x=2$.
\item The parabola is \textbf{above} the $x$-axis when $x<-2$ or $x>2$ (thick part of the axis). Open circles: $\pm2$ give $0$, not $>0$.
\end{enumerate}
\textbf{Solution:} $x<-2$ or $x>2$.
\end{minipage}\hfill
\begin{minipage}[t]{0.26\textwidth}
\vspace{0pt}
\begin{tikzpicture}
\begin{axis}[unitgrid, xmin=-4.5, xmax=4.5, ymin=-5.5, ymax=4.5,
  xtick={-4,...,4}, ytick={-5,...,4}, x=0.4cm, y=0.4cm]
\addplot[very thick, blue, domain=-3:3, samples=60] {x^2-4};

\textbf{B.} A motorcycle is 40 mi ahead of a car. The car goes 60 mph, the motorcycle 40 mph. When is the car ahead?
\begin{enumerate}[label=\arabic*.]
\item Let $x$ = hours. Car: $60x$ mi. Motorcycle: $40x+40$ mi (it \emph{started} 40 mi ahead, so add 40).
\item Car ahead means $60x>40x+40$. Graph $y=60x$ and $y=40x+40$. They intersect at $(2,120)$; after that the $60x$ line is \textbf{above}.
\end{enumerate}
\textbf{Solution:} $x>2$. The car is ahead any time after 2 hours.
